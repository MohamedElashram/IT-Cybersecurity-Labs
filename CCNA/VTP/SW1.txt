! VTP Lab - SW1
! VTP mode: Transparent
! Trunk: Ethernet0/0
! Access ports: Ethernet0/2 and Ethernet0/3 -> VLAN 30

enable
configure terminal
hostname SW1

vtp domain VTP-LAB
vtp mode transparent

vlan 10
 name HR
exit
vlan 20
 name Sales
exit
vlan 30
 name Finance
exit

interface Ethernet0/0
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
exit

interface Ethernet0/2
 switchport mode access
 switchport access vlan 30
 no shutdown
exit

interface Ethernet0/3
 switchport mode access
 switchport access vlan 30
 no shutdown
exit

end
write memory

! Verification:
! show vtp status
! show interfaces trunk
! show vlan brief
